## Применение сохранённой модели

```python
from pathlib import Path
import numpy as np
import pandas as pd
from logistic_regression import FEATURES, prepare, sigmoid

root = Path('.')
model = np.load(root / 'output/model.npz')
data = pd.read_csv(root / 'pima-indians-diabetes.data.csv', header=None,
                   names=FEATURES + ['Outcome'])
parameters = tuple(pd.Series(model[key], index=FEATURES)
                   for key in ['medians', 'means', 'scales'])
X, _ = prepare(data, parameters)
probability = sigmoid(X @ model['weights'])
prediction = (probability >= 0.5).astype(int)
```
